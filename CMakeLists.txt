cmake_minimum_required(VERSION 3.20)
project(ValeoCapture VERSION 0.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "10.15.7" CACHE STRING "Minimum macOS" FORCE)
endif()
option(VALEO_SANITIZERS "Address/undefined sanitizers in host tests" OFF)
option(VALEO_WITH_JPEGLS "Build pinned BSD-licensed CharLS decoder (modern-host download)" ON)
option(VALEO_BUILD_TESTS "Build host verification programs" ON)
option(VALEO_EXPERIMENTAL_READS "Enable fixed diagnostics and allowlisted named identity query pairs" OFF)
option(VALEO_EXPERIMENTAL_CAPTURE "Enable observed raw physical capture for VA11A reference only; requires experimental reads" OFF)
option(VALEO_EXPERIMENTAL_AUXILIARY "Enable statically reviewed auxiliary sequence on experimental SCSI path" OFF)
if(VALEO_EXPERIMENTAL_AUXILIARY AND NOT VALEO_EXPERIMENTAL_CAPTURE)
  message(FATAL_ERROR "Experimental auxiliary requires experimental capture")
endif()
if(VALEO_EXPERIMENTAL_CAPTURE AND NOT VALEO_EXPERIMENTAL_READS)
  message(FATAL_ERROR "Experimental capture requires experimental reads")
endif()
add_library(valeo_core src/protocol.cpp src/replay.cpp src/journal.cpp src/image_writer.cpp src/sbp2.cpp src/receiver.cpp src/identity.cpp src/capture_config.cpp src/raw_validation.cpp src/command_gate.cpp src/raw_pipeline.cpp src/calibration_cache.cpp)
target_include_directories(valeo_core PUBLIC include)
target_compile_options(valeo_core PRIVATE -Wall -Wextra -Wconversion -Wsign-conversion)
if(VALEO_WITH_JPEGLS)
  include(FetchContent)
  if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
  endif()
  set(CHARLS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(CHARLS_BUILD_FUZZ_TEST OFF CACHE BOOL "" FORCE)
  set(CHARLS_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
  set(CHARLS_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(charls
    URL https://codeload.github.com/team-charls/charls/tar.gz/refs/tags/2.4.5
    URL_HASH SHA256=b55c742ef96b5e9d63f813a75d0ba29944302abb5a6ab59756ad61fa017b1dd3)
  FetchContent_MakeAvailable(charls)
  add_library(valeo_jpegls src/jpegls.cpp)
  target_link_libraries(valeo_jpegls PUBLIC valeo_core charls)
  if(VALEO_BUILD_TESTS)
    add_executable(valeo_codec_tests tests/codec_tests.cpp)
    target_link_libraries(valeo_codec_tests PRIVATE valeo_jpegls)
  endif()
endif()
if(VALEO_SANITIZERS)
  target_compile_options(valeo_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(valeo_core PUBLIC -fsanitize=address,undefined)
  if(TARGET charls)
    target_compile_options(charls PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
endif()
if(APPLE)
  add_library(valeo_legacy platform/macos/legacy_transport.cpp)
  target_include_directories(valeo_legacy PRIVATE third_party/apple)
  target_link_libraries(valeo_legacy PUBLIC valeo_core "-framework IOKit" "-framework CoreFoundation")
  target_compile_options(valeo_legacy PRIVATE -Wall -Wextra -Wno-deprecated-declarations)
  if(VALEO_EXPERIMENTAL_READS)
    target_compile_definitions(valeo_legacy PRIVATE VALEO_EXPERIMENTAL_READS=1)
  endif()
  if(VALEO_EXPERIMENTAL_CAPTURE)
    target_compile_definitions(valeo_legacy PRIVATE VALEO_EXPERIMENTAL_CAPTURE=1)
  endif()
  add_library(valeo_scsi platform/macos/scsi_transport.cpp)
  target_include_directories(valeo_scsi SYSTEM PRIVATE third_party/apple)
  target_link_libraries(valeo_scsi PUBLIC valeo_core "-framework IOKit" "-framework CoreFoundation")
  target_compile_options(valeo_scsi PRIVATE -Wall -Wextra -Wno-deprecated-declarations)
  if(VALEO_EXPERIMENTAL_READS)
    target_compile_definitions(valeo_scsi PRIVATE VALEO_EXPERIMENTAL_READS=1)
  endif()
  if(VALEO_EXPERIMENTAL_CAPTURE)
    target_compile_definitions(valeo_scsi PRIVATE VALEO_EXPERIMENTAL_CAPTURE=1)
  endif()
  if(VALEO_EXPERIMENTAL_AUXILIARY)
    target_compile_definitions(valeo_scsi PRIVATE VALEO_EXPERIMENTAL_AUXILIARY=1)
  endif()
  add_library(valeo_host platform/macos/host_transport.cpp)
  target_link_libraries(valeo_host PUBLIC valeo_legacy valeo_scsi)
  add_library(valeo_color platform/macos/color.cpp)
  target_link_libraries(valeo_color PUBLIC valeo_core "-framework CoreGraphics" "-framework CoreFoundation")
  if(VALEO_BUILD_TESTS)
    add_executable(valeo_color_tests tests/color_tests.cpp)
    target_link_libraries(valeo_color_tests PRIVATE valeo_color)
  endif()
  add_library(valeo_bridge platform/macos/bridge.cpp)
  add_library(valeo_session platform/macos/capture_session.cpp platform/macos/capture_export.cpp)
  target_link_libraries(valeo_session PUBLIC valeo_host valeo_color)
  target_compile_options(valeo_session PRIVATE -Wall -Wextra -Wconversion -Wsign-conversion)
  target_link_libraries(valeo_bridge PUBLIC valeo_session)
endif()
add_executable(valeoctl tools/valeoctl.cpp)
if(APPLE)
  target_link_libraries(valeoctl PRIVATE valeo_host)
  target_compile_definitions(valeoctl PRIVATE VALEO_MACOS=1)
else()
  target_link_libraries(valeoctl PRIVATE valeo_core)
endif()
if(VALEO_BUILD_TESTS)
enable_testing()
add_executable(valeo_tests tests/core_tests.cpp)
target_link_libraries(valeo_tests PRIVATE valeo_core)
add_test(NAME core COMMAND valeo_tests)
add_executable(valeo_sbp2_tests tests/sbp2_tests.cpp)
target_link_libraries(valeo_sbp2_tests PRIVATE valeo_core)
add_test(NAME sbp2_completion COMMAND valeo_sbp2_tests)
add_executable(valeo_receiver_tests tests/receiver_tests.cpp)
target_link_libraries(valeo_receiver_tests PRIVATE valeo_core)
add_test(NAME receiver COMMAND valeo_receiver_tests)
add_executable(valeo_identity_tests tests/identity_tests.cpp)
target_link_libraries(valeo_identity_tests PRIVATE valeo_core)
add_test(NAME identity COMMAND valeo_identity_tests)
add_executable(valeo_raw_validation_tests tests/raw_validation_tests.cpp)
target_link_libraries(valeo_raw_validation_tests PRIVATE valeo_core)
add_test(NAME raw_validation COMMAND valeo_raw_validation_tests)
add_executable(valeo_command_gate_tests tests/command_gate_tests.cpp)
target_link_libraries(valeo_command_gate_tests PRIVATE valeo_core)
add_test(NAME command_gate COMMAND valeo_command_gate_tests)
add_executable(valeo_raw_pipeline_tests tests/raw_pipeline_tests.cpp)
target_link_libraries(valeo_raw_pipeline_tests PRIVATE valeo_core)
add_test(NAME raw_pipeline COMMAND valeo_raw_pipeline_tests)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME bundle_profiles COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/bundle_profile_tests.py)
  add_test(NAME bundle_calibration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/calibration_bundle_tests.py)
endif()
if(VALEO_WITH_JPEGLS)
  add_test(NAME jpegls COMMAND valeo_codec_tests ${CMAKE_CURRENT_SOURCE_DIR}/fixtures)
  set_tests_properties(jpegls PROPERTIES TIMEOUT 15)
endif()
if(APPLE)
  add_test(NAME color COMMAND valeo_color_tests)
  add_executable(valeo_capture_export_tests tests/capture_export_tests.cpp)
  target_link_libraries(valeo_capture_export_tests PRIVATE valeo_session)
  add_test(NAME capture_export COMMAND valeo_capture_export_tests)
  set_tests_properties(capture_export PROPERTIES TIMEOUT 300)
  find_program(VALEO_SWIFTC swiftc REQUIRED)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/valeo_session_lifetime_tests
    COMMAND ${VALEO_SWIFTC} -swift-version 5 -O
      ${CMAKE_CURRENT_SOURCE_DIR}/app/service/SessionLifetime.swift
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/session_lifetime_tests.swift
      -o ${CMAKE_CURRENT_BINARY_DIR}/valeo_session_lifetime_tests
    DEPENDS app/service/SessionLifetime.swift tests/session_lifetime_tests.swift)
  add_custom_target(valeo_session_lifetime_test_build ALL DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/valeo_session_lifetime_tests)
  add_test(NAME session_lifetime COMMAND ${CMAKE_CURRENT_BINARY_DIR}/valeo_session_lifetime_tests)
  add_executable(valeo_bridge_state_tests tests/bridge_state_tests.cpp)
  target_include_directories(valeo_bridge_state_tests PRIVATE platform/macos)
  target_link_libraries(valeo_bridge_state_tests PRIVATE valeo_bridge)
  add_test(NAME bridge_state COMMAND valeo_bridge_state_tests)
endif()
find_package(TIFF QUIET)
if(TIFF_FOUND)
  add_executable(valeo_tiff_tests tests/image_tests.cpp)
  target_link_libraries(valeo_tiff_tests PRIVATE valeo_core TIFF::TIFF)
  add_test(NAME independent_tiff COMMAND valeo_tiff_tests)
endif()
endif()
